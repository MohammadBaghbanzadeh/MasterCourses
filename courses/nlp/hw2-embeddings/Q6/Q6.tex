\section{مفاهیم پایه: معیار شباهت کسینوس (بخش \en{۵-۴} کتاب)}

در بخش \textbf{\en{۵-۴}} کتاب (\en{Cosine for measuring similarity})، فرمول شباهت کسینوس به عنوان رایج‌ترین معیار برای اندازه‌گیری شباهت معنایی بین دو بردار کلمه معرفی شده است. این معیار بر اساس زاویه بین دو بردار در فضای برداری عمل می‌کند و مقدار آن بین \(0\) (بدون شباهت) تا \(1\) (بیشترین شباهت) قرار دارد.

فرمول شباهت کسینوس برای دو بردار \(v\) و \(w\) به صورت زیر است (معادله \en{۵-۱۰} کتاب):

\[
\text{\en{cosine}}(v,w) = \frac{v \cdot w}{|v||w|} = \frac{\sum_{i=1}^{N} v_i w_i}{\sqrt{\sum_{i=1}^{N} v_i^2} \cdot \sqrt{\sum_{i=1}^{N} w_i^2}}
\]

ضرب داخلی (دات پروداکت) دو بردار نیز به صورت زیر محاسبه می‌شود:

\[
v \cdot w = v_1 w_1 + v_2 w_2 + \dots + v_N w_N
\]

و اندازه (طول) یک بردار از رابطه زیر به دست می‌آید:

\[
|v| = \sqrt{v_1^2 + v_2^2 + \dots + v_N^2}
\]

\section{داده‌های مسأله}

بردارهای داده شده در صورت سوال:

\begin{align*}
\vec{v}_{\text{\en{king}}} &= [0.8,\; 0.6] \\
\vec{v}_{\text{\en{queen}}} &= [0.6,\; 0.8] \\
\vec{v}_{\text{\en{table}}} &= [0.1,\; 0.05]
\end{align*}

\thickseparator

\section{محاسبه شباهت کسینوس برای هر جفت واژه}

\subsection{گام ۱: محاسبه اندازه (طول) هر بردار}

\noindent\textbf{طول بردار \en{king}:}
\[
|\vec{v}_{\text{\en{king}}}| = \sqrt{(0.8)^2 + (0.6)^2} = \sqrt{0.64 + 0.36} = \sqrt{1.00} = 1.00
\]

\noindent\textbf{طول بردار \en{queen}:}
\[
|\vec{v}_{\text{\en{queen}}}| = \sqrt{(0.6)^2 + (0.8)^2} = \sqrt{0.36 + 0.64} = \sqrt{1.00} = 1.00
\]

\noindent\textbf{طول بردار \en{table}:}
\[
|\vec{v}_{\text{\en{table}}}| = \sqrt{(0.1)^2 + (0.05)^2} = \sqrt{0.01 + 0.0025} = \sqrt{0.0125} \approx 0.1118
\]

\subsection{گام ۲: محاسبه شباهت (\en{king}, \en{queen})}

\noindent\textbf{ابتدا ضرب داخلی:}
\[
\vec{v}_{\text{\en{king}}} \cdot \vec{v}_{\text{\en{queen}}} = (0.8 \times 0.6) + (0.6 \times 0.8) = 0.48 + 0.48 = 0.96
\]

\noindent\textbf{سپس شباهت کسینوس:}
\[
\text{\en{cosine}}(\text{\en{king}}, \text{\en{queen}}) = \frac{0.96}{|\vec{v}_{\text{\en{king}}}| \times |\vec{v}_{\text{\en{queen}}}|} = \frac{0.96}{1.00 \times 1.00} = 0.96
\]

\subsection{گام ۳: محاسبه شباهت (\en{king}, \en{table})}

\noindent\textbf{ضرب داخلی:}
\[
\vec{v}_{\text{\en{king}}} \cdot \vec{v}_{\text{\en{table}}} = (0.8 \times 0.1) + (0.6 \times 0.05) = 0.08 + 0.03 = 0.11
\]

\noindent\textbf{شباهت کسینوس:}
\[
\text{\en{cosine}}(\text{\en{king}}, \text{\en{table}}) = \frac{0.11}{1.00 \times 0.1118} = \frac{0.11}{0.1118} \approx 0.9839
\]

\subsection{گام ۴: محاسبه شباهت (\en{queen}, \en{table})}

\noindent\textbf{ضرب داخلی:}
\[
\vec{v}_{\text{\en{queen}}} \cdot \vec{v}_{\text{\en{table}}} = (0.6 \times 0.1) + (0.8 \times 0.05) = 0.06 + 0.04 = 0.10
\]

\noindent\textbf{شباهت کسینوس:}
\[
\text{\en{cosine}}(\text{\en{queen}}, \text{\en{table}}) = \frac{0.10}{1.00 \times 0.1118} = \frac{0.10}{0.1118} \approx 0.8945
\]

\begin{itemize}
    \item شباهت (\en{king}, \en{queen}) برابر با \(0.96\)
    \item شباهت (\en{king}, \en{table}) برابر با \(0.9839\)
    \item شباهت (\en{queen}, \en{table}) برابر با \(0.8945\)
\end{itemize}

بنابراین \textbf{جفت (\en{king}, \en{table}) با مقدار تقریبی \(0.9839\) بیشترین شباهت کسینوس} را در بین سه جفت دارد.
