\section{الگوریتم پیشنهادی}

\subsection{معماری حل}
قیود چسبندگی سخت یک رابطه هم‌مکانی انتقالی ایجاد می‌کنند. بنابراین ابتدا گراف
$G=(I,E)$ ساخته و مؤلفه‌های همبند آن در زمان $O(|I|+|E|)$ استخراج می‌شوند.
هر مؤلفه به یک اَبَرقلم با وزن مجموع اعضا تبدیل می‌شود. اگر وزن یک مؤلفه از
ظرفیت $C$ بیشتر باشد، نمونه ناموجه است؛ در غیر این صورت مسئله منقبض‌شده دقیقاً
یک \lr{BPP} کلاسیک است و هر جواب آن به یک جواب معتبر برای اقلام اصلی بسط
می‌یابد.

\subsection{جواب اولیه}
دو روش مقداردهی اولیه در پیاده‌سازی واحد پروژه وجود دارد:
\begin{itemize}
  \item \textbf{\lr{FFD}:} مؤلفه‌ها به ترتیب نزولی وزن در نخستین سطل ممکن قرار
  می‌گیرند. این روش در آزمایش اصلی استفاده شده است.
  \item \textbf{\lr{Randomized First Fit}:} ترتیب مؤلفه‌ها و انتخاب مقصد ممکن
  با یک \lr{seed} مشخص تصادفی می‌شود. این روش برای ۳۰ اجرای مستقل تحلیل
  پایداری استفاده شده است.
\end{itemize}
هر دو سازنده فقط جواب موجه تولید می‌کنند.

\subsection{ساختار همسایگی}
تمام حرکات روی مؤلفه‌ها، نه اقلام منفرد، انجام می‌شوند:
\begin{itemize}
  \item \textbf{\lr{MOVE}:} انتقال یک مؤلفه به سطل موجود، مشروط به ظرفیت؛
  \item \textbf{\lr{SWAP}:} مبادله دو مؤلفه از دو سطل متفاوت؛
  \item \textbf{\lr{MERGE}:} ادغام دو سطل در صورت کافی بودن ظرفیت؛
  \item \textbf{\lr{EJECT}:} انتقال یک مؤلفه از سطل چندعضوی به سطل جدید برای
  تنوع‌بخشی جستجو.
\end{itemize}
حرکت \lr{EJECT} موقتاً یک سطل اضافه می‌کند؛ ازاین‌رو در جستجوی عادی انتخاب
نمی‌شود. پس از یک دوره رکود، اگر جواب جاری بیش از یک سطل از بهترین جواب فاصله
نداشته باشد، یک \lr{EJECT} مجاز به‌صورت کنترل‌شده اجرا می‌شود تا ساختار محلی
شکسته شود.

\subsection{تابع ارزیابی، حافظه و توقف}
معیار اصلی کمینه‌سازی تعداد سطل‌هاست. در تساوی معیار اصلی، حرکت با بیشترین
افزایش در
$\sum_b L_b^2$
انتخاب می‌شود؛ این معیار ثانویه سطل‌های پرتر و سطل‌های نزدیک به تخلیه را ترجیح
می‌دهد. تابع \texttt{calculate\_delta} تغییر تعداد سطل‌ها و تغییر مجموع مربعات
بار را بدون ساخت جواب کامل و در $O(1)$ محاسبه می‌کند. تنها حرکت منتخب با یک
کپی خطی از ساختار سطل‌ها اعمال می‌شود.

حافظه تابو به‌جای ذخیره tuple حرکت، شناسه مؤلفه‌های جابه‌جا‌شده را نگه می‌دارد؛
بنابراین حرکت معکوس فوری نیز ممنوع می‌شود. برای پارامتر پایه $T$، دوره تصدی هر
حرکت به‌صورت تصادفی از بازه
$[\max(3,T-2),T+2]$
انتخاب می‌شود. اگر حرکت تابو جوابی بهتر از بهترین جواب تاریخی تولید کند، معیار
تنفس محدودیت را لغو می‌کند. الگوریتم با رسیدن به مقدار مرجع معتبر، رسیدن به
حداکثر تکرار یا عبور از حد رکود متوقف می‌شود.

\subsection{شبه‌کد}
\begin{algorithm}[htbp]
\caption{جستجوی تابو برای بسته‌بندی با قیود چسبندگی}\label{alg:tabu}
\KwIn{اقلام $I$، وزن‌ها $w$، ظرفیت $C$، یال‌ها $E$، پارامترهای $MaxIter,T$}
\KwOut{بهترین بسته‌بندی معتبر $B^*$}
$Components \gets ConnectedComponents(I,E)$\;
\If{$\exists c: W_c>C$}{\Return \lr{Infeasible}}
$B \gets FFD(Components)$; $B^*\gets B$; $TabuUntil\gets\emptyset$\;
\For{$t=1$ \KwTo $MaxIter$}{
  $N(B)\gets MOVE\cup SWAP\cup MERGE\cup EJECT$\;
  \ForEach{$m\in N(B)$}{
    $(\Delta_{bins},\Delta_{sq})\gets CalculateDelta(m)$\;
    حرکت‌های تابو را، مگر در صورت برقراری معیار تنفس، حذف کن\;
  }
  \eIf{رکود دوره‌ای و \lr{EJECT} مجاز وجود دارد}{
    $m^*\gets$ بهترین \lr{EJECT} مجاز\;
  }{
    $m^*\gets\arg\min_m(\Delta_{bins},-\Delta_{sq})$\;
  }
  $B\gets ApplyMove(B,m^*)$\;
  $TabuUntil[c]\gets t+Uniform(\max(3,T-2),T+2)$ برای مؤلفه‌های حرکت\;
  \If{$|B|<|B^*|$}{$B^*\gets B$}
  \If{شرط توقف برقرار است}{\textbf{break}}
}
اعتبار ظرفیت، پوشش مؤلفه‌ها و قیود چسبندگی را بررسی کن\;
\Return $B^*$\;
\end{algorithm}

\subsection{پیچیدگی و پیاده‌سازی}
اگر $m$ تعداد مؤلفه‌ها و $b\le m$ تعداد سطل‌ها باشد، تولید حرکت‌های
\lr{MOVE} از مرتبه $O(mb)$، \lr{SWAP} از مرتبه $O(m^2)$، \lr{MERGE} از
مرتبه $O(b^2)$ و \lr{EJECT} از مرتبه $O(m)$ است. بنابراین هر تکرار در بدترین
حالت $O(m^2)$ و کل جستجو $O(MaxIter\,m^2)$ زمان دارد. چون همسایگی به‌صورت
صریح ساخته می‌شود، حافظه بدترین‌حالت نیز $O(m^2)$ است. اعمال حرکت منتخب
$O(m)$ زمان می‌برد، اما تنها یک‌بار در هر تکرار انجام می‌شود.

برای جلوگیری از اختلاف نسخه‌ها، مدل دقیق، الگوریتم ابتکاری، اعتبارسنجی و تمام
آزمایش‌ها از ماژول واحد \texttt{cohesive\_bin\_packing.py} فراخوانی می‌شوند.
پنج آزمون خودکار نیز contraction، تشخیص ناموجه بودن، صحت دلتا، اعتبار جواب و
تطابق یک نمونه واقعی با جواب دقیق را بررسی می‌کنند.
